function Invoke-VaultSchemaValidation {
  param([string]$Path)

  if (-not (Test-Path -LiteralPath $vaultManagerExe -PathType Leaf)) {
    throw "Vault Manager validator not found: $vaultManagerExe"
  }

  $validationOutput = & $vaultManagerExe validate --vault $Path --scope publish --json
  $validationExitCode = $LASTEXITCODE
  if ($validationExitCode -ne 0) {
    throw "Vault Schema publish validation failed with exit code $validationExitCode.`n$($validationOutput -join "`n")"
  }
  return (($validationOutput -join "`n") | ConvertFrom-Json)
}

function Read-MarkdownFile {
  param([string]$Path)

  $text = Get-Content -LiteralPath $Path -Raw -Encoding UTF8
  $frontmatter = [ordered]@{}
  $body = $text

  if ($text.StartsWith("---`r`n") -or $text.StartsWith("---`n")) {
    $newline = if ($text.StartsWith("---`r`n")) { "`r`n" } else { "`n" }
    $start = 3 + $newline.Length
    $endMarker = $newline + '---' + $newline
    $end = $text.IndexOf($endMarker, $start)
    if ($end -ge 0) {
      $fmText = $text.Substring($start, $end - $start)
      $body = $text.Substring($end + $endMarker.Length)
      $currentKey = $null
      foreach ($line in ($fmText -split "`r?`n")) {
        if ($line -match '^([A-Za-z0-9_-]+):\s*(.*)$') {
          $currentKey = $Matches[1]
          $frontmatter[$currentKey] = $Matches[2].Trim()
        } elseif ($currentKey -and $line -match '^\s+-\s+(.+)$') {
          $existing = [string]$frontmatter[$currentKey]
          if ($existing.StartsWith('[') -and $existing.EndsWith(']')) {
            continue
          }
          $frontmatter[$currentKey] = (($frontmatter[$currentKey], $Matches[1].Trim()) | Where-Object { $_ }) -join ', '
        }
      }
    }
  }

  return [pscustomobject]@{
    Path = $Path
    RelativePath = Get-RelativePath -BasePath $VaultPath -Path $Path
    Frontmatter = $frontmatter
    Body = $body
    Text = $text
  }
}

function Convert-MarkdownPathToDirectoryIndex {
  param([string]$RelativePath)

  if ($RelativePath.EndsWith('/index.md')) {
    return $RelativePath
  }

  $dir = [System.IO.Path]::GetDirectoryName($RelativePath).Replace('\', '/')
  $stem = [System.IO.Path]::GetFileNameWithoutExtension($RelativePath)
  if ([string]::IsNullOrWhiteSpace($dir)) {
    return "$stem/index.md"
  }
  return "$dir/$stem/index.md"
}

function Convert-ContentRelativePath {
  param($Doc)

  $rel = $Doc.RelativePath
  if ($rel -eq 'wiki/index.md') { return 'wiki/index.md' }
  if ($rel.StartsWith('wiki/')) { return Convert-MarkdownPathToDirectoryIndex -RelativePath $rel }
  if ($rel -eq 'notes/software/index.md') { return 'software/guide/index.md' }
  if ($rel.StartsWith('notes/software/')) {
    $id = $Doc.Frontmatter['software_id']
    if ([string]::IsNullOrWhiteSpace($id)) {
      $id = [System.IO.Path]::GetFileNameWithoutExtension($rel)
    }
    return "software/$id/index.md"
  }
  if ($rel.StartsWith('notes/blog/')) {
    $name = [System.IO.Path]::GetFileName($rel)
    if ($name -eq 'index.md') { return 'blog/index.md' }
    $stem = [System.IO.Path]::GetFileNameWithoutExtension($name)
    return "blog/$stem/index.md"
  }
  if ($Doc.Frontmatter['section'] -eq 'blog') {
    $name = [System.IO.Path]::GetFileName($rel)
    $stem = [System.IO.Path]::GetFileNameWithoutExtension($name)
    return "blog/$stem/index.md"
  }
  if ($rel.StartsWith('notes/apps/')) {
    $name = [System.IO.Path]::GetFileName($rel)
    if ($name -eq 'index.md') { return 'app-notes/index.md' }
    $stem = [System.IO.Path]::GetFileNameWithoutExtension($name)
    return "app-notes/$stem/index.md"
  }
  if ($Doc.Frontmatter['section'] -eq 'app-note') {
    $name = [System.IO.Path]::GetFileName($rel)
    $stem = [System.IO.Path]::GetFileNameWithoutExtension($name)
    return "app-notes/$stem/index.md"
  }
  if ($rel.StartsWith('raw/')) {
    $name = [System.IO.Path]::GetFileName($rel)
    $stem = [System.IO.Path]::GetFileNameWithoutExtension($name)
    return "articles/$stem/index.md"
  }
  return Convert-MarkdownPathToDirectoryIndex -RelativePath $rel
}

function Get-ArrayValues {
  param([string]$Value)
  if ([string]::IsNullOrWhiteSpace($Value)) { return @() }
  $trimmed = $Value.Trim()
  if ($trimmed.StartsWith('[') -and $trimmed.EndsWith(']')) {
    return $trimmed.Trim('[', ']') -split ',' | ForEach-Object { $_.Trim().Trim('"', "'") } | Where-Object { $_ }
  }
  if ($trimmed -match ',') {
    return $trimmed -split ',' | ForEach-Object { $_.Trim().Trim('"', "'") } | Where-Object { $_ }
  }
  return @($trimmed.Trim('"', "'"))
}

function Convert-ToQuartzSlugSegment {
  param([string]$Value)

  $slug = $Value.ToLowerInvariant()
  $slug = $slug -replace '%', '-percent'
  $slug = $slug -replace '&', '-and-'
  $slug = $slug -replace '\+', '+'
  $slug = $slug -replace '[\s]+', '-'
  $slug = $slug -replace '[\?#[\]{}<>`"''|\\^~;:,@!$*()=]+', ''
  $slug = $slug -replace '-+', '-'
  $slug = $slug.Trim('-')
  return $slug
}

function Get-ImageReferences {
  param([string]$Text)
  $refs = [System.Collections.Generic.List[string]]::new()

  foreach ($match in [regex]::Matches($Text, '!\[\[([^\]|]+)(?:\|[^\]]+)?\]\]')) {
    [void]$refs.Add($match.Groups[1].Value.Trim())
  }
  foreach ($match in [regex]::Matches($Text, '!\[[^\]]*\]\(([^)]+)\)')) {
    $target = $match.Groups[1].Value.Trim()
    if ($target -notmatch '^(https?:)?//') {
      [void]$refs.Add(($target -split '\s+"' | Select-Object -First 1).Trim())
    }
  }

  return $refs
}

function Resolve-AssetPath {
  param($Doc, [string]$Reference)

  $cleanRef = [Uri]::UnescapeDataString(($Reference -split '#')[0]).Trim()
  if ([string]::IsNullOrWhiteSpace($cleanRef)) { return $null }
  $ext = [System.IO.Path]::GetExtension($cleanRef).ToLowerInvariant()
  if ($imageExtensions -notcontains $ext) { return $null }

  $candidates = @(
    (Join-Path -Path (Split-Path -Parent $Doc.Path) -ChildPath $cleanRef),
    (Join-Path -Path $VaultPath -ChildPath $cleanRef),
    (Join-Path -Path (Join-Path $VaultPath 'assets') -ChildPath $cleanRef)
  )

  foreach ($candidate in $candidates) {
    if (Test-Path -LiteralPath $candidate -PathType Leaf) {
      return (Resolve-Path -LiteralPath $candidate).Path
    }
  }

  $fileName = [System.IO.Path]::GetFileName($cleanRef)
  $found = Get-ChildItem -LiteralPath (Join-Path $VaultPath 'assets') -Recurse -File -ErrorAction SilentlyContinue |
    Where-Object { $_.Name -eq $fileName } |
    Select-Object -First 1
  if ($found) { return $found.FullName }
  return $null
}

function Convert-PublicHtmlToDirectoryIndexes {
  param([string]$PublicPath)

  if (-not (Test-Path -LiteralPath $PublicPath)) { return }

  $resolvedPublic = (Resolve-Path -LiteralPath $PublicPath).Path
  $htmlFiles = Get-ChildItem -LiteralPath $PublicPath -Recurse -File -Filter '*.html' |
    Where-Object { $_.Name -ne 'index.html' }

  foreach ($file in $htmlFiles) {
    $resolvedFile = (Resolve-Path -LiteralPath $file.FullName).Path
    if (-not $resolvedFile.StartsWith($resolvedPublic, [System.StringComparison]::OrdinalIgnoreCase)) {
      throw "Refusing to rewrite HTML outside public: $resolvedFile"
    }

    $targetDir = Join-Path $file.DirectoryName $file.BaseName
    $resolvedTargetDirParent = (Resolve-Path -LiteralPath $file.DirectoryName).Path
    if (-not $resolvedTargetDirParent.StartsWith($resolvedPublic, [System.StringComparison]::OrdinalIgnoreCase)) {
      throw "Refusing to create clean URL outside public: $targetDir"
    }

    if (-not (Test-Path -LiteralPath $targetDir)) {
      New-Item -ItemType Directory -Path $targetDir | Out-Null
    }

    $html = Get-Content -LiteralPath $file.FullName -Raw -Encoding UTF8
    $html = [regex]::Replace($html, '(?<attr>\b(?:href|src)=["''])(?<url>(?!https?:|mailto:|javascript:|#|/)[^"'']+)(?<end>["''])', {
      param($match)
      return $match.Groups['attr'].Value + '../' + $match.Groups['url'].Value + $match.Groups['end'].Value
    })
    $html = $html -replace 'fetch\("\.\./static/', 'fetch("../../static/'
    Set-Content -LiteralPath (Join-Path $targetDir 'index.html') -Value $html -Encoding UTF8
    Remove-Item -LiteralPath $file.FullName -Force
  }
}

function Convert-ToPublicIndexSlug {
  param([string]$Slug)

  if ([string]::IsNullOrWhiteSpace($Slug)) { return '/' }

  $clean = $Slug.Replace('\', '/').Trim()
  $clean = $clean.TrimStart('/')
  if ($clean -eq 'index') { return '/' }
  if ($clean.EndsWith('/index')) {
    $clean = $clean.Substring(0, $clean.Length - '/index'.Length)
  }
  $clean = $clean.TrimEnd('/')
  if ([string]::IsNullOrWhiteSpace($clean)) { return '/' }
  return $clean
}

function Normalize-ContentIndexForCleanUrls {
  param([string]$PublicPath)

  $indexPath = Join-Path $PublicPath 'static\contentIndex.json'
  if (-not (Test-Path -LiteralPath $indexPath -PathType Leaf)) { return }

  $raw = Get-Content -LiteralPath $indexPath -Raw -Encoding UTF8
  if ([string]::IsNullOrWhiteSpace($raw)) { return }

  $index = $raw | ConvertFrom-Json -AsHashtable
  $normalized = [ordered]@{}

  foreach ($key in $index.Keys) {
    $cleanKey = Convert-ToPublicIndexSlug -Slug ([string]$key)
    if ($normalized.Contains($cleanKey)) {
      continue
    }

    $entry = $index[$key]
    if ($entry -is [System.Collections.IDictionary]) {
      $entry['slug'] = $cleanKey
      if ($entry.Contains('links') -and $entry['links']) {
        $entry['links'] = @($entry['links'] | ForEach-Object { Convert-ToPublicIndexSlug -Slug ([string]$_) })
      }
    }

    $normalized[$cleanKey] = $entry
  }

  $normalized | ConvertTo-Json -Depth 20 -Compress | Set-Content -LiteralPath $indexPath -Encoding UTF8
}

function Convert-PublicMarkdown {
  param($Doc)

  $text = $Doc.Text
  if (-not ($text.StartsWith("---`r`n") -or $text.StartsWith("---`n"))) {
    return $text
  }

  $newline = if ($text.StartsWith("---`r`n")) { "`r`n" } else { "`n" }
  $start = 3 + $newline.Length
  $endMarker = $newline + '---' + $newline
  $end = $text.IndexOf($endMarker, $start)
  if ($end -lt 0) { return $text }

  $fmLines = [System.Collections.Generic.List[string]]::new()
  $sourceBlock = $false
  foreach ($line in ($text.Substring($start, $end - $start) -split "`r?`n")) {
    if ($line -match '^published:\s*(?:null)?\s*$') {
      continue
    }
    if ($line -match '^sources:\s*$') {
      $sourceBlock = $true
      continue
    }
    if ($sourceBlock -and $line -match '^\s+-\s+') {
      continue
    }
    if ($sourceBlock -and $line -match '^[A-Za-z0-9_-]+:') {
      $sourceBlock = $false
    }
    if (-not $sourceBlock) {
      [void]$fmLines.Add($line)
    }
  }

  $body = $text.Substring($end + $endMarker.Length)
  $bodyLines = [System.Collections.Generic.List[string]]( $body -split "`r?`n" )
  $outBody = [System.Collections.Generic.List[string]]::new()
  $skipSourceSection = $false
  foreach ($line in $bodyLines) {
    if ($line -match '^##\s+来源\s*$') {
      $skipSourceSection = $true
      continue
    }
    if ($skipSourceSection -and $line -match '^##\s+') {
      $skipSourceSection = $false
    }
    if (-not $skipSourceSection) {
      [void]$outBody.Add($line)
    }
  }

  $publicBody = ($outBody -join "`n")
  $publicBody = [regex]::Replace($publicBody, '`(?:daily|output|raw)/[^`]+?\.md`', '未公开来源材料')
  $publicBody = [regex]::Replace($publicBody, '(?m)^-\s*(?:daily|output|raw)/.+?\.md\s*$', '')
  $publicBody = [regex]::Replace($publicBody, '\[([^\]]+)\]\((?:\.\./)?(?:原始材料|daily|output|raw)/[^)]+?\.md(?:\s+"[^"]*")?\)', '$1')
  $publicBody = [regex]::Replace($publicBody, '(?m)^\[[^\]]+\]:\s+(?:\.\./)?(?:原始材料|daily|output|raw)/.+?\.md(?:\s+".*?")?\s*$', '')
  $publicBody = [regex]::Replace($publicBody, '\[\[([^\]|#]+)(#[^\]|]+)?(\|[^\]]+)?\]\]', {
    param($match)
    $target = $match.Groups[1].Value.Trim()
    $anchor = $match.Groups[2].Value
    $alias = $match.Groups[3].Value
    $targetExt = [System.IO.Path]::GetExtension(($target -split '#')[0]).ToLowerInvariant()
    if ($imageExtensions -contains $targetExt) { return $match.Value }

    $targetKey = Convert-WikilinkTargetToKey -Target $target
    if ($script:publishedLinkTargets -and $script:publishedLinkTargets.ContainsKey($targetKey)) {
      return '[[' + $script:publishedLinkTargets[$targetKey] + $anchor + $alias + ']]'
    }
    return $match.Value
  })

  if ($Doc.Frontmatter['section'] -eq 'wiki' -and $Doc.RelativePath -ne 'wiki/index.md') {
    $kind = switch ($Doc.Frontmatter['type']) {
      'concept' { '概念' }
      'entity' { '实体' }
      'topic' { '主题' }
      default { '条目' }
    }
    $status = switch ($Doc.Frontmatter['status']) {
      'mature' { '成熟' }
      'stub' { '小条目' }
      default { [string]$Doc.Frontmatter['status'] }
    }
    if ([string]::IsNullOrWhiteSpace($status)) { $status = '未标注' }
    $tags = (Get-ArrayValues -Value ([string]$Doc.Frontmatter['tags']) | Select-Object -First 4) -join ' · '
    if ([string]::IsNullOrWhiteSpace($tags)) { $tags = '无' }
    $summary = [System.Net.WebUtility]::HtmlEncode([string]$Doc.Frontmatter['summary'])
    $title = [System.Net.WebUtility]::HtmlEncode([string]$Doc.Frontmatter['title'])
    $updated = [System.Net.WebUtility]::HtmlEncode([string]$Doc.Frontmatter['updated'])

    $wikiFrame = @"
<div class="wiki-entry">
<aside class="wiki-infobox">
<div class="wiki-infobox-title">$title</div>
<div class="wiki-facts">
<div><span>类型</span><strong>$kind</strong></div>
<div><span>状态</span><strong>$status</strong></div>
<div><span>更新</span><strong>$updated</strong></div>
<div><span>标签</span><strong>$tags</strong></div>
</div>
</aside>
<p class="wiki-lead">$summary</p>
</div>

"@
    $publicBody = $wikiFrame + $publicBody
  }

  return "---`n" + ($fmLines -join "`n") + "`n---`n" + ($publicBody.TrimStart("`r", "`n"))
}

function Get-SlugFromOutput {
  param([string]$Output)
  if ($Output -match '/index\.md$') {
    return ($Output -replace '/index\.md$', '/')
  }
  return ($Output -replace '\.md$', '')
}

function Get-HrefFromOutput {
  param([string]$Output)
  $slug = Get-SlugFromOutput -Output $Output
  $trailingSlash = $slug.EndsWith('/')
  $segments = $slug.TrimEnd('/') -split '/'
  $safeSegments = $segments | ForEach-Object { Convert-ToQuartzSlugSegment -Value $_ }
  $href = ($safeSegments -join '/')
  if ($trailingSlash) { $href += '/' }
  return $href
}

function Convert-WikilinkTargetToKey {
  param([string]$Target)

  $clean = ($Target -split '#')[0].Trim().Trim('/')
  $clean = $clean -replace '\.md$', ''
  if ([string]::IsNullOrWhiteSpace($clean)) { return '' }

  $segments = $clean -split '/'
  return (($segments | ForEach-Object { Convert-ToQuartzSlugSegment -Value $_ }) -join '/')
}

function Get-SortedEntries {

function New-EntryList {
  param($Items, [string]$ExtraClass = '')
  $className = 'entry-row'
  if (-not [string]::IsNullOrWhiteSpace($ExtraClass)) {
    $className = "$className $ExtraClass"
  }
  return ((Get-SortedEntries -Items $Items | ForEach-Object {

    $summary = [System.Net.WebUtility]::HtmlEncode([string]$_.summary)

  }) -join "`n")
}

function Convert-ToDataValue {
  param([string]$Value)
  return (Convert-ToQuartzSlugSegment -Value $Value)
}

function New-ExpandableEntryList {
  param($Items, [string]$ContainerClass, [int]$VisibleCount = 12, [string]$EmptyText = '暂无内容。')

  $sorted = @(Get-SortedEntries -Items $Items)
  if ($sorted.Count -eq 0) { return $EmptyText }

  $visible = @($sorted | Select-Object -First $VisibleCount)
  $hidden = @($sorted | Select-Object -Skip $VisibleCount)
  $visibleRows = New-EntryList -Items $visible
  $html = @"
<div class="$ContainerClass">
$visibleRows
</div>
"@

  if ($hidden.Count -gt 0) {
    $hiddenRows = New-EntryList -Items $hidden -ExtraClass 'is-collapsed-entry'
    $html += @"

<details class="collection-more">
<summary>查看更多 $($hidden.Count) 篇</summary>
<div class="$ContainerClass collection-more-list">
$hiddenRows
</div>
</details>
"@
  }

  return $html
}

function Get-AppNoteRelatedSoftware {
  param($AppNote, [string[]]$SoftwareIds)

  $explicit = [string]$AppNote.frontmatter['related_software']
  if (-not [string]::IsNullOrWhiteSpace($explicit)) {
    return (Convert-ToQuartzSlugSegment -Value $explicit)
  }

  $fileName = ($AppNote.source -split '/')[-1]
  $stem = [System.IO.Path]::GetFileNameWithoutExtension($fileName)
  foreach ($id in ($SoftwareIds | Sort-Object Length -Descending)) {
    if ($stem -eq $id -or $stem.StartsWith("$id-", [System.StringComparison]::OrdinalIgnoreCase)) {
      return $id
    }
  }

  return $null
}

function Add-SectionAfterFirstHeading {
  param([string]$Path, [string]$Section)
  if ([string]::IsNullOrWhiteSpace($Section)) { return }

  $text = Get-Content -LiteralPath $Path -Raw -Encoding UTF8
  $lines = [System.Collections.Generic.List[string]]($text -split "`r?`n")
  for ($i = 0; $i -lt $lines.Count; $i++) {
    if ($lines[$i] -match '^#\s+') {
      $lines.Insert($i + 1, $Section.Trim("`r", "`n"))
      Set-Content -LiteralPath $Path -Value ($lines -join "`n") -Encoding UTF8
      return
    }
  }

  Set-Content -LiteralPath $Path -Value ($Section.Trim("`r", "`n") + "`n" + $text) -Encoding UTF8
}

$publishedOutputs = @{}
foreach ($item in $contentDocs) {
  $slug = (Get-HrefFromOutput -Output $item.output).Trim('/')
  Add-PublishedOutputKey -Map $publishedOutputs -Key $slug
  Add-PublishedOutputKey -Map $publishedOutputs -Key (($slug -split '/')[-1])
  Add-PublishedOutputKey -Map $publishedOutputs -Key (Convert-WikilinkTargetToKey -Target $item.source)
  Add-PublishedOutputKey -Map $publishedOutputs -Key ([System.IO.Path]::GetFileNameWithoutExtension($item.source))
}

foreach ($doc in $published) {
  foreach ($match in [regex]::Matches($doc.Text, '\[\[([^\]|#]+)(?:#[^\]|]+)?(?:\|[^\]]+)?\]\]')) {
    $target = $match.Groups[1].Value.Trim()
    if ([string]::IsNullOrWhiteSpace($target)) { continue }
    $targetExt = [System.IO.Path]::GetExtension(($target -split '#')[0]).ToLowerInvariant()
    if ($imageExtensions -contains $targetExt) { continue }
    $targetKey = Convert-WikilinkTargetToKey -Target $target
    if (-not $publishedOutputs.ContainsKey($targetKey)) {
      $possibleName = Convert-ToQuartzSlugSegment -Value ([System.IO.Path]::GetFileNameWithoutExtension($target))
      if (-not $publishedOutputs.ContainsKey($possibleName)) {
        [void]$brokenLinks.Add([pscustomobject]@{ page = $doc.RelativePath; link = $target })
      }
    }
  }
}

$blockedUnpublished = @($docs | Where-Object { -not (Test-PublishTrue -Doc $_) -and (($_.RelativePath.StartsWith('daily/')) -or ($_.RelativePath.StartsWith('output/')) -or ($_.RelativePath.StartsWith('raw/'))) } | ForEach-Object {
  [pscustomobject]@{ path = $_.RelativePath; reason = 'not publish:true' }
})

  sensitiveHits = $sensitiveHits
  sensitiveTermHits = $sensitiveTermHits
  brokenLinks = $brokenLinks
  placeholderHits = $placeholderHits
  unlinkedAppNotes = $unlinkedAppNotes
  missingImages = $missingImages
  copiedImages = $copiedImages
  privateImages = $privateImages
  canPublish = ($missingFields.Count -eq 0 -and $blockedFiles.Count -eq 0 -and $sensitiveHits.Count -eq 0 -and $brokenLinks.Count -eq 0 -and $placeholderHits.Count -eq 0 -and $unlinkedAppNotes.Count -eq 0 -and $missingImages.Count -eq 0 -and $privateImages.Count -eq 0)
}
